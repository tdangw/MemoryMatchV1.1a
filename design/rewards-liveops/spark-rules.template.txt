    function rewardWallet(n,o) {
      let sameDay = n.day == o.day;
      let sameWeek = n.week == o.week;
      let spent = n.ticketSpent - o.ticketSpent;
      let dayBonus = sameDay && n.daily.claimed == o.daily.claimed ? 0 : sameDay ? (!o.daily.claimed.hasAny(['bonus']) && n.daily.claimed.hasAny(['bonus']) ? 1 : 0)
        : (!o.daily.claimed.hasAny(['bonus']) && o.daily.wins >= 3 && o.daily.matches >= 30 && o.daily.noHint >= 1 ? 1 : 0);
      let weekBonus = sameWeek && n.weekly.claimed == o.weekly.claimed ? 0 : sameWeek ? (!o.weekly.claimed.hasAny(['bonus']) && n.weekly.claimed.hasAny(['bonus']) ? 2 : 0)
        : (!o.weekly.claimed.hasAny(['bonus']) && o.weekly.wins >= 12 && o.weekly.matches >= 120 && o.weekly.noHint >= 3 && o.weekly.visits >= 3 ? 2 : 0);
      return n.ticketEarned >= o.ticketEarned && spent >= 0 && spent <= 1
        && n.tickets == o.tickets + n.ticketEarned - o.ticketEarned + dayBonus + weekBonus - spent;
    }
